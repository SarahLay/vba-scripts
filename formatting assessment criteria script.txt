' Paste into a Module in the VBA editor (Alt+F11 -> Insert -> Module)
Option Explicit

Sub FormatAcceptanceCriteria()
    Dim rng As Range, cell As Range
    Dim raw As String, norm As String
    Dim lines() As String, i As Long
    Dim groups As Collection, curr As String
    Dim out As String, outOffset As Long
    
    outOffset = 1 ' write output to column to the right; set 0 to overwrite originals
    If TypeName(Selection) <> "Range" Then Exit Sub
    Set rng = Selection.Columns(1).Cells
    
    For Each cell In rng
        If Trim(CStr(cell.Value)) <> "" Then
            raw = CStr(cell.Value)
            norm = Replace(raw, vbCrLf, vbLf)
            norm = Replace(norm, vbCr, vbLf)
            lines = Split(norm, vbLf)
            
            Set groups = New Collection
            curr = ""
            For i = LBound(lines) To UBound(lines)
                Dim t As String
                t = Trim(lines(i))
                If t <> "" Then
                    If curr = "" Then
                        curr = t
                    Else
                        curr = curr & vbLf & t
                    End If
                Else
                    If curr <> "" Then
                        groups.Add curr
                        curr = ""
                    End If
                End If
            Next i
            If curr <> "" Then groups.Add curr
            
            If groups.Count > 0 Then
                out = ""
                Dim gIdx As Long
                For gIdx = 1 To groups.Count
                    Dim grp As String
                    grp = groups(gIdx)
                    Dim glines() As String
                    glines = Split(grp, vbLf)
                    
                    out = out & "<div>" & vbLf
                    Dim j As Long
                    For j = LBound(glines) To UBound(glines)
                        Dim lineTxt As String
                        lineTxt = Trim(glines(j))
                        If lineTxt <> "" Then
                            If j = LBound(glines) Then
                                Dim pos As Long
                                pos = InStr(lineTxt, ":")
                                If pos > 0 Then
                                    Dim prefix As String, rest As String
                                    prefix = Left(lineTxt, pos) ' includes colon
                                    rest = Trim(Mid(lineTxt, pos + 1))
                                    If rest <> "" Then
                                        out = out & "<p><b>" & EscapeHtml(prefix) & "</b> " & EscapeHtml(rest) & "</p>" & vbLf
                                    Else
                                        out = out & "<p><b>" & EscapeHtml(prefix) & "</b></p>" & vbLf
                                    End If
                                    GoTo NextLineInGroup
                                End If
                            End If
                            ' Default: bold first word
                            out = out & BoldFirstWordP(lineTxt) & vbLf
                        End If
NextLineInGroup:
                    Next j
                    out = out & "</div>" & vbLf & "<br>" & vbLf & vbLf
                Next gIdx
                ' no outer wrapper; groups are individual <div> blocks separated by <br>
                
                cell.Offset(0, outOffset).Value = out
            End If
        End If
    Next cell
End Sub

Private Function BoldFirstWordP(ByVal txt As String) As String
    Dim parts() As String
    parts = Split(txt, " ")
    If UBound(parts) >= 1 Then
        BoldFirstWordP = "<p><b>" & EscapeHtml(parts(0)) & "</b> " & EscapeHtml(Mid(txt, Len(parts(0)) + 2)) & "</p>"
    Else
        BoldFirstWordP = "<p><b>" & EscapeHtml(txt) & "</b></p>"
    End If
End Function

' Minimal HTML-escape for < & >
Private Function EscapeHtml(s As String) As String
    s = Replace(s, "&", "&amp;")
    s = Replace(s, "<", "&lt;")
    s = Replace(s, ">", "&gt;")
    EscapeHtml = s
End Function
